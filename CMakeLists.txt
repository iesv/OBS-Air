cmake_minimum_required(VERSION 3.28)
project(OBS-Air VERSION 2.0.0 LANGUAGES C CXX)

if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8 OR (NOT WIN32 AND NOT APPLE))
  message(FATAL_ERROR "Build for Windows x64 or macOS arm64")
endif()
if(WIN32 AND NOT MSVC)
  message(FATAL_ERROR "Windows builds require MSVC")
endif()
if(APPLE AND NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
  message(FATAL_ERROR "macOS builds support Apple Silicon arm64 only")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
set(DEPS "${CMAKE_SOURCE_DIR}/.deps" CACHE PATH "Dependency cache")
list(PREPEND CMAKE_PREFIX_PATH "${DEPS}/qt" "${DEPS}/obs-deps")
find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
if(Qt6_VERSION VERSION_LESS 6.8 OR NOT Qt6_VERSION VERSION_LESS 6.9)
  message(FATAL_ERROR "Build with Qt 6.8.x")
endif()
set(OPENSSL_ROOT_DIR "${DEPS}/openssl")
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)

file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
file(WRITE "${CMAKE_BINARY_DIR}/generated/obsconfig.h"
  "#pragma once\n#define OBS_DATA_PATH \"data\"\n#define OBS_PLUGIN_PATH \"obs-plugins/64bit\"\n#define OBS_PLUGIN_DESTINATION \"obs-plugins/64bit\"\n#define OBS_RELEASE_CANDIDATE 0\n#define OBS_BETA 0\n")
set(OBS_HEADERS "${DEPS}/obs-studio/libobs")
add_library(obs_sdk INTERFACE)
target_include_directories(obs_sdk INTERFACE "${OBS_HEADERS}" "${CMAKE_BINARY_DIR}/generated")
if(WIN32)
  execute_process(COMMAND "${CMAKE_AR}" /nologo
    "/def:${CMAKE_SOURCE_DIR}/cmake/obs.def" "/out:${CMAKE_BINARY_DIR}/obs.lib" /machine:x64
    COMMAND_ERROR_IS_FATAL ANY)
  target_link_libraries(obs_sdk INTERFACE "${CMAKE_BINARY_DIR}/obs.lib")
  add_library(air_threads STATIC "${DEPS}/obs-studio/deps/w32-pthreads/pthread.c")
  target_compile_definitions(air_threads PUBLIC PTW32_STATIC_LIB PRIVATE __CLEANUP_C PTW32_BUILD)
  target_include_directories(air_threads PUBLIC "${DEPS}/obs-studio/deps/w32-pthreads")
else()
  find_file(OBS_LIBRARY libobs PATHS
    "${DEPS}/obs-runtime/OBS.app/Contents/Frameworks/libobs.framework" NO_DEFAULT_PATH REQUIRED)
  target_link_libraries(obs_sdk INTERFACE "${OBS_LIBRARY}")
  find_package(Threads REQUIRED)
  add_library(air_threads INTERFACE)
  target_link_libraries(air_threads INTERFACE Threads::Threads)
endif()

set(PLIST "${CMAKE_SOURCE_DIR}/third_party/libplist")
add_library(air_plist STATIC
  ${PLIST}/src/base64.c ${PLIST}/src/bplist.c ${PLIST}/src/bytearray.c
  ${PLIST}/src/hashtable.c ${PLIST}/src/jplist.c ${PLIST}/src/jsmn.c
  ${PLIST}/src/oplist.c ${PLIST}/src/out-default.c ${PLIST}/src/out-limd.c
  ${PLIST}/src/out-plutil.c ${PLIST}/src/plist.c ${PLIST}/src/ptrarray.c
  ${PLIST}/src/time64.c ${PLIST}/src/xplist.c
  ${PLIST}/libcnary/node.c ${PLIST}/libcnary/node_list.c)
target_include_directories(air_plist PUBLIC "${PLIST}/include" PRIVATE
  "${PLIST}/libcnary/include" "${PLIST}/src" "${PLIST}" "${CMAKE_SOURCE_DIR}/cmake")
target_compile_definitions(air_plist PUBLIC LIBPLIST_STATIC PRIVATE HAVE_CONFIG_H=1)

set(UX "${CMAKE_SOURCE_DIR}/third_party/uxplay/lib")
file(GLOB PLAYFAIR CONFIGURE_DEPENDS "${UX}/playfair/*.c")
file(GLOB LLHTTP CONFIGURE_DEPENDS "${UX}/llhttp/*.c")
add_library(air_protocol STATIC ${PLAYFAIR} ${LLHTTP}
  ${UX}/byteutils.c ${UX}/compat.c ${UX}/crypto.c ${UX}/dnssd.c
  ${UX}/fairplay_playfair.c ${UX}/http_request.c ${UX}/http_response.c ${UX}/httpd.c
  ${UX}/logger.c ${UX}/mirror_buffer.c ${UX}/netutils.c ${UX}/pairing.c ${UX}/raop.c
  ${UX}/raop_buffer.c ${UX}/raop_ntp.c ${UX}/raop_rtp.c ${UX}/raop_rtp_mirror.c
  ${UX}/srp.c ${UX}/utils.c)
target_include_directories(air_protocol PUBLIC "${UX}" PRIVATE "${UX}/playfair" "${UX}/llhttp")
target_compile_definitions(air_protocol PUBLIC PLIST_210 PLIST_230 OPENSSL_API_COMPAT=0x10101000L
  PRIVATE LLHTTP_STATIC)
target_link_libraries(air_protocol PUBLIC air_threads air_plist OpenSSL::Crypto)
if(WIN32)
  target_sources(air_protocol PRIVATE ${UX}/mdnsd/dnssd_mdnsd.c ${UX}/mdnsd/mdnsd.c)
  target_compile_definitions(air_protocol PUBLIC WIN32 WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(air_protocol PUBLIC ws2_32 iphlpapi)
else()
  target_sources(air_protocol PRIVATE ${UX}/dns_sd/dns_sd.c)
  # Bonjour's DNS-SD client API is provided by macOS libSystem.
endif()

add_library(air_ffmpeg INTERFACE)
target_include_directories(air_ffmpeg INTERFACE "${DEPS}/obs-deps/include")
target_link_directories(air_ffmpeg INTERFACE "${DEPS}/obs-deps/lib")
target_link_libraries(air_ffmpeg INTERFACE avcodec avutil swscale)
add_library(obs-air MODULE src/plugin.cpp src/source.cpp src/decoders.cpp)
target_compile_definitions(obs-air PRIVATE OBS_AIR_VERSION="${PROJECT_VERSION}")
target_link_libraries(obs-air PRIVATE obs_sdk Qt6::Core air_protocol air_ffmpeg)
set_target_properties(obs-air PROPERTIES PREFIX "")
if(WIN32)
  target_compile_options(obs-air PRIVATE /W4 /permissive- /utf-8)
  foreach(target IN ITEMS air_threads air_plist air_protocol obs-air)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS _CRT_NONSTDC_NO_DEPRECATE)
  endforeach()
else()
  target_compile_options(obs-air PRIVATE -Wall -Wextra)
  set_target_properties(obs-air PROPERTIES SUFFIX "" BUILD_WITH_INSTALL_RPATH TRUE
    INSTALL_RPATH "@loader_path/../Frameworks;@executable_path/../Frameworks")
endif()

include(CTest)
if(BUILD_TESTING)
  add_executable(media-tests tests/media-test.cpp)
  target_include_directories(media-tests PRIVATE src)
  add_test(NAME media COMMAND media-tests)
  add_executable(decoder-tests tests/decoder-test.cpp src/decoders.cpp)
  target_compile_definitions(decoder-tests PRIVATE LIBOBS_EXPORTS FIXTURE_DIR="${CMAKE_SOURCE_DIR}/tests/fixtures")
  target_include_directories(decoder-tests PRIVATE src "${OBS_HEADERS}" "${CMAKE_BINARY_DIR}/generated")
  target_link_libraries(decoder-tests PRIVATE air_ffmpeg)
  if(WIN32)
    target_compile_options(decoder-tests PRIVATE /utf-8)
  endif()
  add_test(NAME decoder-last-frame COMMAND decoder-tests)
  add_test(NAME decoder-stress COMMAND decoder-tests --stress)
  set_tests_properties(decoder-stress PROPERTIES TIMEOUT 180 LABELS stress)
  if(WIN32)
    set_tests_properties(decoder-last-frame decoder-stress PROPERTIES ENVIRONMENT "PATH=${DEPS}/obs-deps/bin;$ENV{PATH}")
  else()
    set_tests_properties(decoder-last-frame decoder-stress PROPERTIES ENVIRONMENT "DYLD_LIBRARY_PATH=${DEPS}/obs-deps/lib")
  endif()
  add_executable(protocol-tests tests/protocol-test.cpp)
  target_link_libraries(protocol-tests PRIVATE air_protocol)
  add_test(NAME protocol-lifecycle COMMAND protocol-tests)
  set_tests_properties(protocol-lifecycle PROPERTIES TIMEOUT 30)
  add_executable(password-tests tests/password-test.cpp)
  target_link_libraries(password-tests PRIVATE air_protocol)
  add_test(NAME password-lifecycle COMMAND password-tests)
  set_tests_properties(password-lifecycle PROPERTIES TIMEOUT 60)
  add_executable(slides-tests tests/slides-test.cpp src/decoders.cpp)
  target_compile_definitions(slides-tests PRIVATE LIBOBS_EXPORTS OBS_AIR_VERSION="${PROJECT_VERSION}")
  target_link_libraries(slides-tests PRIVATE obs_sdk Qt6::Core air_protocol air_ffmpeg)
  add_test(NAME slides-loopback COMMAND slides-tests)
  set_tests_properties(slides-loopback PROPERTIES TIMEOUT 120)
  if(WIN32)
    target_compile_options(slides-tests PRIVATE /utf-8)
    set_tests_properties(slides-loopback PROPERTIES ENVIRONMENT
      "PATH=${DEPS}/obs-runtime/bin/64bit;${DEPS}/obs-deps/bin;$ENV{PATH}")
  else()
    set_tests_properties(slides-loopback PROPERTIES ENVIRONMENT
      "DYLD_LIBRARY_PATH=${DEPS}/obs-runtime/OBS.app/Contents/Frameworks:${DEPS}/obs-deps/lib")
  endif()
endif()

if(WIN32)
  set(PLUGIN_BIN "obs-studio/plugins/obs-air/bin/64bit")
  install(TARGETS obs-air LIBRARY DESTINATION "${PLUGIN_BIN}")
  install(FILES "${DEPS}/obs-deps/bin/avcodec-61.dll" "${DEPS}/obs-deps/bin/avutil-59.dll"
    "${DEPS}/obs-deps/bin/swscale-8.dll" "${DEPS}/obs-deps/bin/swresample-5.dll"
    "${DEPS}/obs-deps/bin/libx264-164.dll" "${DEPS}/obs-deps/bin/zlib.dll" DESTINATION "${PLUGIN_BIN}")
  install(DIRECTORY data/ DESTINATION "obs-studio/plugins/obs-air/data")
  install(FILES README.md LICENSE THIRD_PARTY.md scripts/install_obs-air.bat DESTINATION ".")
  install(FILES "${CMAKE_BINARY_DIR}/THIRD_PARTY_LICENSES.txt" DESTINATION ".")
  set(CPACK_GENERATOR ZIP)
  set(CPACK_PACKAGE_FILE_NAME "OBS-Air-${PROJECT_VERSION}-windows-x64")
  include(CPack)
endif()
